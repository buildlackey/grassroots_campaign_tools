interface IProgressReport {
    int totalToProcess
    int processedSoFar
    boolean  done()
    boolean  failed()
    boolean  succeeded()
    Json     details
}

class AbstractChunkedTaskProcessor {
    attributes
    int totalToProcess
    int chunkSize) => AbstractChunkedTaskProcessor (singleton)

    // methods

    createInstance(totalToProcess,  chunkSize) => AbstractChunkedTaskProcessor (singleton)

    abstract processChunk(rangeStart, rangeEnd): void
    abstract progressReporter(): IProgressReport

    void process() {
        int startIndex = 0
        while(true) {
            IProgressReport progress <= progressReporter()
            if (progress.done()) 
                break;
            int end = Max(startIndex+step, totalToProcess)
            processChunk(startIndex, endIndex)
        } 

    }
}
